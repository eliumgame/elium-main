# UTF-8
#
# Ressource VERSIONINFO Windows embarquee dans Elium.exe par PyInstaller
# (EXE(version=...) dans installer/elium.spec). Un binaire sans cette ressource
# ressemble davantage a un dropper generique aux yeux des heuristiques ML de
# Windows Defender (cf. recherche false-positive Wacatac.B!ml) ; un vrai
# CompanyName/FileDescription/ProductName est un signal de legitimite peu
# couteux a fournir, meme s'il ne suffit pas seul a lever une detection ML.
#
# Les deux triplets ci-dessous sont synchronises par installer/stamp_version.py
# (meme mecanisme que les autres emplacements de version du projet) : NE PAS
# les editer a la main sans repasser par ce script ensuite.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(4, 5, 4, 0),
    prodvers=(4, 5, 4, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          u"040c04b0",
          [
            StringStruct(u"CompanyName", u"Elium Authors"),
            StringStruct(
              u"FileDescription",
              u"Elium — éditeur de documents .elium chiffrés et signés, local-first",
            ),
            StringStruct(u"FileVersion", u"4.5.4"),
            StringStruct(u"InternalName", u"Elium"),
            StringStruct(u"LegalCopyright", u"© Elium Authors"),
            StringStruct(u"OriginalFilename", u"Elium.exe"),
            StringStruct(u"ProductName", u"Elium"),
            StringStruct(u"ProductVersion", u"4.5.4"),
          ],
        )
      ]
    ),
    VarFileInfo([VarStruct(u"Translation", [1036, 1200])]),
  ],
)
