/**
 * ATLAS — capture de TOUS les écrans / états / thèmes / largeurs + métriques de
 * mise en page automatiques (débordements, textes tronqués, cibles trop petites).
 * Fichier jetable (zz*) : sort dans $ATLAS_OUT, ne fait partie d'aucune CI.
 */
const __dirname = path.dirname(fileURLToPath(import.meta.url));
test.use({ actionTimeout: 4000 });
const OUT = process.env.ATLAS_OUT ?? path.join(__dirname, "..", "atlas-out");
const BASE = "http://localhost:3100/";

type Issue = { shot: string; kind: string; detail: string };
const issues: Issue[] = [];
const index: { shot: string; ok: boolean; note?: string }[] = [];

const METRICS = () => {
  const out: { kind: string; detail: string }[] = [];
  const vw = window.innerWidth;
  const doc = document.documentElement;
  if (doc.scrollWidth > vw + 1) out.push({ kind: "overflow-x", detail: `page ${doc.scrollWidth}px > fenêtre ${vw}px` });
  const vis = (e: Element) => {
    const r = e.getBoundingClientRect();
    const cs = getComputedStyle(e);
    return r.width > 0 && r.height > 0 && cs.visibility !== "hidden" && cs.display !== "none" && cs.opacity !== "0";
  };
  const name = (e: Element) =>
    e.tagName.toLowerCase() + (e.id ? "#" + e.id : "") + "." + String((e as HTMLElement).className || "").toString().split(/\s+/).slice(0, 3).join(".");
  let clipped = 0;
  let small = 0;
  let outside = 0;
  for (const e of Array.from(document.querySelectorAll("body *"))) {
    if (!vis(e)) continue;
    const el = e as HTMLElement;
    const r = el.getBoundingClientRect();
    const cs = getComputedStyle(el);
    // texte coupé (ellipsis ou overflow hidden avec contenu plus large)
    if (el.scrollWidth > el.clientWidth + 2 && (cs.overflowX === "hidden" || cs.textOverflow === "ellipsis") && el.children.length === 0 && (el.textContent ?? "").trim().length > 1) {
      if (clipped++ < 6) out.push({ kind: "texte-tronque", detail: `${name(el)} « ${(el.textContent ?? "").trim().slice(0, 40)} » (${el.scrollWidth}>${el.clientWidth})` });
    }
    // cibles interactives trop petites (< 24px) — WCAG 2.2 AA 2.5.8
    if (/^(button|a|select|input)$/i.test(el.tagName) && (el as HTMLInputElement).type !== "hidden" && (r.width < 24 || r.height < 24) && r.width > 4) {
      if (small++ < 6) out.push({ kind: "cible-petite", detail: `${name(el)} ${Math.round(r.width)}×${Math.round(r.height)} « ${(el.getAttribute("aria-label") || el.textContent || "").trim().slice(0, 30)} »` });
    }
    // éléments qui sortent de l'écran à droite (hors scroll interne)
    if (r.right > vw + 2 && r.width < vw && !el.closest("[style*='overflow'], .elx-ribbon__body, .pdfx-ribbon__body, table, pre")) {
      if (outside++ < 6) out.push({ kind: "hors-ecran", detail: `${name(el)} right=${Math.round(r.right)} > ${vw}` });
    }
  }
  return out;
};

function flush() {
  fs.mkdirSync(OUT, { recursive: true });
  fs.writeFileSync(path.join(OUT, "index.json"), JSON.stringify(index, null, 1));
  fs.writeFileSync(path.join(OUT, "issues.json"), JSON.stringify(issues, null, 1));
}

async function shot(page: Page, dir: string, name: string, opts: { full?: boolean } = {}) {
  const file = `${dir}/${name}.png`;
  fs.mkdirSync(path.dirname(path.join(OUT, file)), { recursive: true });
  try {
    await page.waitForTimeout(350);
    await page.screenshot({ path: path.join(OUT, file), fullPage: !!opts.full });
    const m = await page.evaluate(METRICS);
    for (const i of m) issues.push({ shot: file, ...i });
    index.push({ shot: file, ok: true, note: m.length ? `${m.length} signal(s)` : undefined });
    flush();
  } catch (e) {
    index.push({ shot: file, ok: false, note: String(e).slice(0, 120) });
    flush();
  }
}

async function step(label: string, fn: () => Promise<unknown>) {
  try {
    await fn();
    return true;
  } catch (e) {
    index.push({ shot: `STEP:${label}`, ok: false, note: String(e).slice(0, 140) });
    flush();
    return false;
  }
}

async function newPage(ctx: BrowserContext) {
  const page = await ctx.newPage();
  await page.addInitScript(() => {
    const w = window as unknown as Record<string, unknown>;
    delete w.showSaveFilePicker;
    delete w.showOpenFilePicker;
  });
  return page;
}

const goHome = async (p: Page) => {
  await p.goto(BASE);
  await p.waitForTimeout(1800);
};

    ["Récents", "04-recents"],

    await goHome(p);

    await p.getByRole("button", { name: "Palette de commandes" }).click();

    await p.keyboard.press("Escape");

const VIEWPORTS: { id: string; w: number; h: number; scheme: "light" | "dark"; sets: string[] }[] = [
  { id: "bureau-clair", w: 1440, h: 900, scheme: "light", sets: ["home", "settings", "doc", "sheet", "slides", "pdf", "others"] },
  { id: "bureau-sombre", w: 1440, h: 900, scheme: "dark", sets: ["home", "settings", "doc", "sheet", "slides", "pdf", "others"] },
  { id: "portable-1024", w: 1024, h: 768, scheme: "light", sets: ["home", "settings", "doc", "sheet", "slides", "pdf", "others"] },
  { id: "tablette-768", w: 768, h: 1024, scheme: "light", sets: ["home", "settings", "doc", "sheet", "slides", "pdf", "others"] },
  { id: "mobile-390", w: 390, h: 844, scheme: "light", sets: ["home", "settings", "doc", "sheet", "slides", "pdf", "others"] },
  { id: "mobile-sombre-390", w: 390, h: 844, scheme: "dark", sets: ["home", "settings", "doc"] },
];

for (const vp of VIEWPORTS) {
  test(`atlas ${vp.id}`, async ({ browser }) => {
    test.setTimeout(560_000);
    const ctx = await browser.newContext({ viewport: { width: vp.w, height: vp.h }, colorScheme: vp.scheme });
    const fns: Record<string, (p: Page, d: string) => Promise<void>> = { home: homeAndWorkspace, settings, doc: documents, sheet, slides, pdf, others };
    for (const s of vp.sets) {
      const p = await newPage(ctx);
      await step(`${vp.id}/${s}`, () => fns[s]!(p, vp.id));
      await p.close();
    }
    await ctx.close();
  });
}

test.afterAll(async () => {
  fs.mkdirSync(OUT, { recursive: true });
  fs.writeFileSync(path.join(OUT, "index.json"), JSON.stringify(index, null, 1));
  fs.writeFileSync(path.join(OUT, "issues.json"), JSON.stringify(issues, null, 1));
});
